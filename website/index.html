using System;using System.Collections.Generic;using System.Drawing;using System.Linq;using System.Windows.Forms;namespace memory_project{    public partial class game : Form    {   //je punten        int points = 0;        public game()        {            InitializeComponent();            Timer.Text = "0";            flipBackTimer.Interval = 500;            flipBackTimer.Tick += FlipBackTimer_Tick;            StartGame();        }        // start game        private void StartGame()        {            flipBackTimer.Stop();            //zodat de restart button niet gedisabled is            button25.Enabled = true;            timer1.Start();            points = 0;            label1.Text = "Points: 0";            Timer.Text = "0 seconds";            firstClicked = null;            secondClicked = null;            List<Color> shuffledColors = BuildColorPairs();            int index = 0;            foreach (Button btn in GetAllButtons(this))            {                if (btn != button25 && btn.Name.StartsWith("button"))                {                    btn.BackColor = Color.White;                    btn.Tag = shuffledColors[index];                    btn.Enabled = true;                    index++;                }            }        }        // dit is waar er kleuren worden gepaird en geschud        private List<Color> BuildColorPairs()        {            List<Color> colors = new List<Color>();            if (clicked == null || clicked.Tag == null)                return;            if (clicked == firstClicked)                return;            clicked.BackColor = (Color)clicked.Tag;            if (firstClicked == null)            {                firstClicked = clicked; # Begin van het spel# Print een bericht voor invoer van het woord# Speler 1 voert een woord in# Speler 2 is nu aan de beurt en moet het woord raden.# Woord klaarzetten# Streepjes voor elke letter van het woord.# Zet het aantal fouten dat mag.# Letters raden# Speler 2 kiest een letter en voert dit in.# Als de letter al geraden is, voer een andere in.# Als de letter goed is, vul hem in het woord in.# Als de letter fout is, teken een deel van de galg en tel de fout op.# Checken of het spel voorbij is# Als alle letters geraden zijn → speler 2 wint.# Als het maximum aantal fouten bereikt is → woordbedenker wint.# Einde# Laat zien wie gewonnen heeft en wat het woord was.def draw_hangman(mistakes):    stages = [                    # 50 legen lijnen zodat player 2 niet het geheimen woord ziet    print(" Player 2's beurt ")    guessed_letters = []  #letters die al geraden zijn    mistakes = 0        #fouten die al gemaakt zijn    max_mistakes = 7    #7 fouten maximaal    visible = ["_" if c.isalpha() else c for c in secret_word] #maakt van de niet geraden letters een _, als je een letter raad scant hij                                                               #het woord en zet hem neer op de lijn code, als hij er bijstaat word hij neergezet anders is het een fout    while mistakes < max_mistakes and "_" in visible:  #zorgt ervoor dat letters invisable blijven totdat ze geraden zijn        draw_hangman(mistakes)        print("\nWoord: " + " ".join(visible))        print(f"fouten: {mistakes}/{max_mistakes}")        print("geraden letters:", ", ".join(guessed_letters))        guess = input("gok een letter: ").lower().strip()    # gokt een letter en haalt spaties en hoofdletters weg        if len(guess) != 1 or not guess.isalpha():   # als het meer als 1 letter is is het een fout            print("1 letter tegelijk aub.")            continue        if guess in guessed_letters:         # geen dubbele letters raden            print("deze letter heb je al geraden.")            continue        guessed_letters.append(guess)   # als letter geraden is dan        if guess in secret_word:            # print goed en laat de letter zien            print("goed!")            for i, letter in enumerate(secret_word):                if letter == guess:                    visible[i] = guess        else:                  # anders print fout en +1 bij mistakes            print("fout")            mistakes += 1    draw_hangman(mistakes)    if "_" not in visible:          # als er geen _ meer zijn print je hebt het woord geraden en wat het woord was nog een keer        print("\nje hebt het woord geraden!:", secret_word)    else:        print("\nverloren, het woord was:", secret_word)       # meer dan 7 fouten heb je verlorenif __name__ == "__main__":    hangman()
